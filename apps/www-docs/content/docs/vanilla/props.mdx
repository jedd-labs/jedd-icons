The vanilla package exposes the shared styling model as `createElement`
options, plus a couple specific to imperative rendering and the `createIcons`
helper.

## createElement options

<TypeTable
  type={{
    size: { type: "number | string", default: "24", description: "Sets both width and height." },
    color: { type: "string", default: '"currentColor"', description: "Sets the stroke color." },
    strokeWidth: { type: "number | string", default: "2", description: "Stroke width." },
    absoluteStrokeWidth: { type: "boolean", default: "false", description: "Keep a constant visual stroke width at any size." },
    class: { type: "string", description: "Class names set on the root <svg>. createElement adds no jedd classes of its own." },
    attrs: { type: "Record<string, string | number>", description: "Extra attributes set on the root <svg>." },
  }}
/>

## createIcons options

<TypeTable
  type={{
    icons: { type: "Record<string, IconNode>", default: "{}", description: "Map of icon names to icon data. The name must match the data-jedd value." },
    attrs: { type: "CreateElementOptions", default: "{}", description: "Default options applied to every replaced icon." },
    nameAttr: { type: "string", default: '"data-jedd"', description: "The attribute used to find placeholders." },
    watch: { type: "boolean", default: "false", description: "Keep replacing icons as new placeholders are added to the DOM." },
  }}
/>

import {
  createElement,
  createIcons,
  defaultAttributes,
  variantDefaults,
  resolveStrokeWidth,
} from "@jedd-icons/core";

  IconNode,
  CreateElementOptions,
  CreateIconsOptions,
} from "@jedd-icons/core";
